type M = Record<string, unknown>;

function measure(): M {
  const out: M = {};
  const q = (id: string) => document.querySelector<HTMLElement>(`[data-s="${id}"]`);
  const rect = (el: Element) => el.getBoundingClientRect().toJSON();

  // ۱) جای rt نسبت به پایه — بالای پایه است؟ افقی مرکز؟
  for (const id of ['s1', 's2a', 's3bdi', 's3span', 's4', 's6a', 's6b', 's7under', 's8pattern']) {
    const sec = q(id);
    if (!sec) continue;
    const ruby = sec.querySelector('ruby');
    const rt = ruby?.querySelector('rt');
    const baseSpan = ruby?.querySelector('[data-nb]');
    if (!ruby || !rt) continue;
    const rr = rect(ruby), tr = rect(rt);
    type R = { top: number; left: number; width: number; height: number };
    let baseRect: R | undefined;
    if (baseSpan) {
      baseRect = rect(baseSpan);
    } else {
      const base = ruby.firstChild;
      if (!base) continue;
      const br = document.createRange();
      br.selectNodeContents(base);
      baseRect = Array.from(br.getClientRects())[0]?.toJSON();
    }
    if (!baseRect) continue;
    out[id] = {
      rtAboveBase: tr.top < baseRect.top,
      rtTopDelta: +(baseRect.top - tr.top).toFixed(1),
      rtHeight: +(tr.height).toFixed(1),
      baseHeight: +(baseRect.height).toFixed(1),
      rtCenterOverBase: baseRect.width
        ? +((tr.left + tr.width / 2 - (baseRect.left + baseRect.width / 2))).toFixed(1)
        : null,
      rtWiderThanBase: +(tr.width - baseRect.width).toFixed(1),
      rubyLineHeightRatio: +(rr.height / baseRect.height).toFixed(2),
    };
  }

  // ۲) هم‌ترازی خط پایهٔ ruby با متن فارسی اطراف
  const s1 = q('s1');
  if (s1) {
    const fa = s1.querySelector('[data-fa]');
    const ruby = s1.querySelector('ruby');
    if (fa && ruby) {
      const fr = rect(fa), rr = rect(ruby);
      out.s1Baseline = {
        faTop: +fr.top.toFixed(1), faBottom: +fr.bottom.toFixed(1),
        rubyTop: +rr.top.toFixed(1), rubyBottom: +rr.bottom.toFixed(1),
        // کف جعبهٔ ruby با کف جعبهٔ متن فارسی یکی است؟
        bottomDelta: +(rr.bottom - fr.bottom).toFixed(1),
        rubyBoxGrowsUpward: rr.top < fr.top,
      };
    }
  }

  // ۳) تداخل خطی در line-height تنگ — rt خط n با متن خط n-1 برخورد می‌کند؟
  for (const id of ['s5lh14', 's5lh20']) {
    const sec = q(id);
    if (!sec) continue;
    const rts = Array.from(sec.querySelectorAll('rt'));
    const spans = Array.from(sec.querySelectorAll('[data-nb]')); // همسایه‌های غیرروبی
    let overlaps = 0;
    const detail: string[] = [];
    for (const rt of rts) {
      const a = rt.getBoundingClientRect();
      for (const sp of spans) {
        if (sp.contains(rt) || rt.contains(sp)) continue;
        const b = sp.getBoundingClientRect();
        const inter = !(a.right < b.left || b.right < a.left || a.bottom < b.top || b.bottom < a.top);
        if (inter) {
          overlaps++;
          detail.push(`${(rt.textContent || '').slice(0, 6)}×${(sp.textContent || '').slice(0, 6)}`);
        }
      }
    }
    out[id] = { rtCount: rts.length, neighborOverlaps: overlaps, detail: detail.slice(0, 8) };
  }

  // ۴) انتخاب متن — آیا روبی وارد selection می‌شود؟
  const s2 = q('s2a');
  if (s2) {
    const p = s2.querySelector('p');
    if (p) {
      const range = document.createRange();
      range.selectNodeContents(p);
      const sel = window.getSelection();
      sel?.removeAllRanges();
      sel?.addRange(range);
      const text = sel?.toString() || '';
      out.selection = {
        includesFurigana: text.includes('にほんご') || text.includes('きょう'),
        includesBase: text.includes('日本語') || text.includes('今日'),
        sampleOrder: text.replace(/\s+/g, ' ').slice(0, 80),
      };
      sel?.removeAllRanges();
    }
  }

  // ۵) rp — در مرورگر پشتیبان‌دار مخفی است؟
  const rp = document.querySelector('rp');
  if (rp) {
    const cs = getComputedStyle(rp);
    out.rp = { display: cs.display, contentHidden: cs.display === 'none' };
  }

  // ۶) مقادیر پیش‌فرض محاسبه‌شدهٔ CSS روبی در این مرورگر
  const ruby = document.querySelector('ruby');
  if (ruby) {
    const cs = getComputedStyle(ruby);
    out.computedDefaults = {
      rubyPosition: cs.rubyPosition || (cs as unknown as Record<string, string>)['ruby-position'],
      rubyAlign: cs.rubyAlign || (cs as unknown as Record<string, string>)['ruby-align'],
      writingMode: cs.writingMode,
      direction: cs.direction,
      userAgent: navigator.userAgent,
    };
  }

  // ۷) رندر دوزبانهٔ الگوی نهایی — وجود خط پایه/شکست
  const s8 = q('s8pattern');
  if (s8) {
    const island = s8.querySelector('.island');
    if (island) {
      const cs = getComputedStyle(island);
      out.s8island = { direction: cs.direction, unicodeBidi: cs.unicodeBidi };
    }
  }
  return out;
}
